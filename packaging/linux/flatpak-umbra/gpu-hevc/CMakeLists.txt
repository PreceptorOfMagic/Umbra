# Production-facing development DLLs. x265 is linked only for configuration
# syntax/context initialization; frame pixels never enter x265_encoder_encode.
option(BUILD_HOST_DLLS "Build Windows D3D11 GPU encoder libraries" OFF)
option(GPU_HEVC_RETRY_TRACE "Trace rate attempts in smoke executables only" OFF)
if(BUILD_HOST_DLLS)
  if(NOT MSVC)
    message(FATAL_ERROR "Host DLLs currently require MSVC and matching static x265 libraries")
  endif()
  set(X265_SOURCE "" CACHE PATH "x265 checkout pinned in README")
  foreach(depth 8 10)
    set(X265_BUILD_${depth} "" CACHE PATH "Static-CRT x265 build for ${depth} bit")
    set(xbuild "${X265_BUILD_${depth}}")
    if(NOT EXISTS "${xbuild}/Release/x265-static.lib" OR NOT EXISTS "${xbuild}/x265_config.h")
      message(FATAL_ERROR "Missing Release static x265 ${depth}-bit build")
    endif()
    add_library(aurora_gpu_hevc${depth} SHARED src/aurora_gpu_hevc.cpp)
    add_executable(gpu_rate_picture_smoke${depth} tests/gpu_rate_picture_smoke.cpp)
    if(GPU_HEVC_RETRY_TRACE)
      target_compile_definitions(gpu_rate_picture_smoke${depth} PRIVATE GPU_HEVC_RETRY_TRACE=1)
    endif()
    foreach(host_target aurora_gpu_hevc${depth} gpu_rate_picture_smoke${depth})
    set_target_properties(${host_target} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_include_directories(${host_target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${OPENCL_INCLUDE_DIR}" "${xbuild}")
    target_include_directories(${host_target} SYSTEM PRIVATE "${X265_SOURCE}/source" "${X265_SOURCE}/source/common" "${X265_SOURCE}/source/encoder")
    target_compile_options(${host_target} PRIVATE /W4 /WX /O2 /external:W0)
    if(depth EQUAL 10)
      set(high_bit_depth 1)
    else()
      set(high_bit_depth 0)
    endif()
    target_compile_definitions(${host_target} PRIVATE CL_TARGET_OPENCL_VERSION=120 X265_DEPTH=${depth} HIGH_BIT_DEPTH=${high_bit_depth} X265_NS=x265)
    target_link_libraries(${host_target} PRIVATE "${xbuild}/Release/x265-static.lib" d3d11 dxgi advapi32 d3dcompiler)
    endforeach()
  endforeach()
endif()

option(BUILD_LINUX_HOST_LIBS "Build Linux OpenCL GPU encoder libraries" OFF)
if(BUILD_LINUX_HOST_LIBS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Linux host libraries require a Linux toolchain")
  endif()
  set(X265_SOURCE "" CACHE PATH "x265 checkout pinned in README")
  foreach(depth 8 10)
    set(X265_BUILD_${depth} "" CACHE PATH "PIC static x265 build for ${depth} bit")
    set(xbuild "${X265_BUILD_${depth}}")
    if(NOT EXISTS "${xbuild}/libx265.a" OR NOT EXISTS "${xbuild}/x265_config.h")
      message(FATAL_ERROR "Missing PIC static x265 ${depth}-bit build")
    endif()
    add_library(aurora_gpu_hevc${depth} SHARED src/aurora_gpu_hevc_linux.cpp)
    target_include_directories(aurora_gpu_hevc${depth} PRIVATE src "${OPENCL_INCLUDE_DIR}" "${xbuild}")
    target_include_directories(aurora_gpu_hevc${depth} SYSTEM PRIVATE "${X265_SOURCE}/source" "${X265_SOURCE}/source/common" "${X265_SOURCE}/source/encoder")
    if(depth EQUAL 10)
      set(high_bit_depth 1)
    else()
      set(high_bit_depth 0)
    endif()
    target_compile_definitions(aurora_gpu_hevc${depth} PRIVATE CL_TARGET_OPENCL_VERSION=120 X265_DEPTH=${depth} HIGH_BIT_DEPTH=${high_bit_depth} X265_NS=x265)
    target_compile_options(aurora_gpu_hevc${depth} PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
    # Each depth owns its x265 globals; ELF interposition must not bind either
    # to the host's libx265 or to the other depth loaded in the same process.
    target_link_options(aurora_gpu_hevc${depth} PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic
      "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/linux.exports")
    set_property(TARGET aurora_gpu_hevc${depth} APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/linux.exports")
    target_link_libraries(aurora_gpu_hevc${depth} PRIVATE "${xbuild}/libx265.a" ${CMAKE_DL_LIBS} pthread)
    install(TARGETS aurora_gpu_hevc${depth} LIBRARY DESTINATION lib/umbra/gpu-hevc)
  endforeach()
  install(DIRECTORY kernels DESTINATION lib/umbra/gpu-hevc)
  install(FILES COPYING COPYING-GPL-3 DESTINATION share/licenses/umbra-gpu-hevc)
endif()
